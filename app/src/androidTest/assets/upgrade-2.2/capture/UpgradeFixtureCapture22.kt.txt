/**
 * Not a test: run once at tag `v2.2` to capture the 2.2 data directory that `UpgradeFrom22Test` reads at head.
 * Every value goes through 2.2's own stores into `externalCacheDir/upgrade-2.2/{files,no_backup,cache}` (the app's
 * real directories stay untouched). Tokens go through an identity cipher into `plaintext/`: a Keystore key cannot be
 * shipped, so the head test encrypts them with the frozen 2.2 envelope. All values are synthetic.
 */
@RunWith(AndroidJUnit4::class)
class UpgradeFixtureCapture22 {

    private val target = InstrumentationRegistry.getInstrumentation().targetContext
    private val root = File(checkNotNull(target.externalCacheDir), "upgrade-2.2")
    private val files = File(root, "files")
    private val noBackup = File(root, "no_backup")
    private val cache = File(root, "cache")
    private val plaintext = File(root, "plaintext")
    private val context: Context = object : ContextWrapper(target) {
        override fun getFilesDir() = files
        override fun getNoBackupFilesDir() = noBackup
        override fun getCacheDir() = cache
        override fun getApplicationContext() = this
    }
    private val clock = Clock.fixed(CAPTURED_AT, ZoneOffset.UTC)
    private val gson = NetworkModule.provideGson(NetworkModule.provideWidgetsClient(MyItmo(), "https://localhost/"))

    @Test
    fun capture() = runBlocking {
        root.deleteRecursively()
        check(files.mkdirs() && noBackup.mkdirs() && cache.mkdirs() && plaintext.mkdirs())

        val barsTokens = BarsTokenStore(File(plaintext, "bars_tokens.enc"), IdentityCipher)
        capturePreferences(barsTokens)
        captureTokens(barsTokens)
        captureFileStores()
        captureWidgetAndCaches()
        captureSpoiler()
        captureDebugOverrides()
        captureDiagnostics()
    }

    private suspend fun capturePreferences(barsTokens: BarsTokenStore) {
        val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
        val preferences = PreferenceDataStoreFactory.create(
            scope = scope,
            produceFile = { File(files, "datastore/app_preferences.preferences_pb") }
        )
        val settings = AppSettingsStorage(preferences)
        settings.setCustomServicesEnabled(true)
        settings.setScheduleSportAutoSignEnabled(true)
        settings.setScheduleChangesEnabled(false)
        // The per-format keys differ from the legacy shared keys below, so a read shows which one won.
        settings.setCompactWidgetNextLessonEarlyEnabled(true)
        settings.setCompactWidgetTeacherHidden(false)
        settings.setFullWidgetTeacherHidden(true)
        settings.setFullWidgetPastLessonsHidden(false)
        settings.setFullWidgetTomorrowEnabled(false)
        settings.setCompactWidgetTextSize(WidgetTextSize.LARGE)
        settings.setFullWidgetTextSize(WidgetTextSize.EXTRA_LARGE)
        // 2.2 only reads these four (written by 2.0 and 2.1); no 2.2 setter exists.
        preferences.edit {
            it[booleanPreferencesKey("widget_forward_scheduling_enabled")] = false
            it[booleanPreferencesKey("widget_hide_teacher_enabled")] = true
            it[booleanPreferencesKey("widget_hide_previous_lessons_enabled")] = true
            it[booleanPreferencesKey("widget_future_schedule_enabled")] = true
        }
        settings.setQrDynamicColorsEnabled(false)
        settings.setQrSpoilerEnabled(false)
        settings.setQrSpoilerAnimationType(QrAnimationType.FADE)
        settings.setSportSignHideTeacherSelectorEnabled(false)
        settings.setSportSignHideTimeSelectorEnabled(false)
        settings.setMyItmoMarksEnabled(false)
        settings.setSheetMarksEnabled(false)
        settings.setBarsMarksEnabled(false)
        settings.setBarsLoginPrompt(BarsLoginPrompt.SHOWN)
        settings.setBackgroundWorkHintShown()
        settings.setQrTileAdded(true)
        settings.dismissHomeHint("WIDGETS")
        settings.dismissHomeHint("SERVICES")
        settings.setHomeCardHidden("SPORT", true)
        settings.setHomeCardHidden("FRIEND_REQUESTS", true)
        settings.setDemoActive(true)

        val utility = UtilityStorage(preferences, appVersionName = "2.2")
        utility.setRegisteredFirebaseToken("upgrade22-registered-fcm-token", ownerIsu = ISU)
        utility.setFirebaseToken("upgrade22-fcm-token")
        utility.setLastUpdateTimestamp(CAPTURED_MS - 3_600_000)
        utility.setLessonWidgetStyleChanged(false)
        utility.setSkippedVersion("2.1.1")
        utility.setVersionNotificationTimestamp(CAPTURED_MS - 7_200_000)
        utility.setOnboardingCompleted(true)

        DataStoreSubjectBindingStore(preferences).apply {
            put(1001, 2001)
            put(1002, 2002)
        }
        check(BarsPreferenceRepositoryImpl(barsTokens, preferences, settings).setEnabled(true) is AppResult.Success)
        QrWidgetStateStoreImpl(preferences).setState(42, QrWidgetState.VISIBLE)
        DataStoreFriendSelectionHistory(preferences).apply {
            record(100002)
            record(100003)
        }

        assertEquals(40, preferences.data.first().asMap().size)
        scope.cancel()
    }

    private fun captureTokens(barsTokens: BarsTokenStore) {
        MyItmoStorage(File(plaintext, "myitmo_tokens.enc"), IdentityCipher, clock).replaceWithTokens(
            SessionTokens(
                accessToken = "upgrade22-access-token",
                accessExpiresInSeconds = HUNDRED_YEARS_SECONDS,
                refreshToken = "upgrade22-refresh-token",
                refreshExpiresInSeconds = HUNDRED_YEARS_SECONDS,
                idToken = ID_TOKEN
            )
        )
        barsTokens.install(ISU, "Bearer upgrade22-bars-session")
    }

    private fun captureSpoiler() {
        val source = File(checkNotNull(target.externalCacheDir), "upgrade22-spoiler-source.png")
        source.outputStream().use { check(solid(64, Color.rgb(200, 60, 120)).compress(Bitmap.CompressFormat.PNG, 100, it)) }
        assertTrue(CustomSpoilerManager(context).saveCustomSpoiler(Uri.fromFile(source)))
        source.delete()
    }

    private fun captureDebugOverrides() {
        FileSportLessonTemplateStore(File(noBackup, "debug/sport_lesson_templates")).setEnabled(true)
        FileSportScoreOverrideStore(File(noBackup, "debug/sport_score_override"))
            .setOverride(SportScoreOverride(attendances = 7, bonus = 3))
        FileAcademicTimeOverrideStore(File(noBackup, "debug/academic_date_override"))
            .setOverrideDate(LocalDate.parse("2026-10-12"))
    }

    private suspend fun captureDiagnostics() {
        val diagnostics = FileAppDiagnostics(File(files, "diagnostics"), clock)
        diagnostics.warn("UpgradeCapture", "Synthetic warning of the 2.2 fixture", null)
        diagnostics.awaitWrites()
        assertEquals(1, File(files, "diagnostics/log.jsonl").readLines().size)
    }

    private fun solid(size: Int, color: Int): Bitmap = createBitmap(size, size).apply { eraseColor(color) }

    private object IdentityCipher : TokenCipher {
        override fun encrypt(value: String) = value
        override fun decrypt(value: String) = value
    }

    private object FixedTime : AcademicTimeProvider {
        override val zoneId: ZoneId = ZoneId.of("Europe/Moscow")
        override fun today(): LocalDate = LocalDate.now(Clock.fixed(CAPTURED_AT, zoneId))
        override fun now(): OffsetDateTime = OffsetDateTime.ofInstant(CAPTURED_AT, zoneId)
    }

    private object SignedInTokens : dev.alllexey.itmowidgets.core.session.SessionTokenStore {
        override fun hasRefreshToken() = true
        override fun getIdToken() = ID_TOKEN
        override fun replaceWithRefreshToken(refreshToken: String) = Unit
        override fun replaceWithTokens(tokens: SessionTokens) = Unit
        override fun clearTokens() = Unit
    }

        const val HUNDRED_YEARS_SECONDS = 3_153_600_000L
        const val ISU = 123456
        const val SUBJECT = "Тестовая дисциплина"
        const val HALF = "2026/2027-1"
        const val PERIOD = "2026-1"
        const val LINK_ID = "00000000-0000-4000-8000-000000000022"
        const val QR_HEX = "0f1e2d3c4b5a69788796a5b4c3d2e1f0"

            .joinToString(".", postfix = ".signature") {
                Base64.getUrlEncoder().withoutPadding().encodeToString(it.toByteArray(Charsets.UTF_8))
            }
    }
